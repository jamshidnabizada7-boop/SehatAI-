    53→   107→/** Detects forbidden dose amounts — units (mg/ml/mcg/g), tablet/pill/goli

    56→   110→export function hasDosePattern(text: string): boolean {
    57→   111→  let m: RegExpExecArray | null;
    58→   112→  // unit-based doses: "500 mg", "5 ml", "2 g", "100 mcg"
    59→   113→  const unitRe = /\b\d+(?:\.\d+)?\s*(?:mg|ml|mcg|g)\b/gi;
    60→   114→  while ((m = unitRe.exec(text)) !== null) {
    61→   115→    const windowStart = Math.max(0, m.index - 70);
    62→   116→    const window = text.slice(windowStart, m.index + m[0].length + 70).toLowerCase();
    63→   117→    // ORS standard preparation instructions and glucose readings may mention amounts
    64→   118→    if (/ors|oral rehydration|او آر ایس|sachet|packet|litre|liter/.test(window)) continue;
    65→   119→    if (/mg\s*\/?\s*dl|blood sugar|sugar|glucose|shakar|شوگر|reading|level/.test(window)) continue;
    66→   120→    return true;
    67→   121→  }
    68→   122→  // countable dosage forms: "2 tablets", "3 goli", "1 capsule", "2 drops", "half tablet"
    69→   123→  const formRe = /\b(?:\d+(?:\.\d+)?|one|two|three|four|five|half|do|teen|char|paanch)\s*(?:tablets?|pills?|goli|goliyan|goliyaan|capsules?|drops?|spoonfuls?|chammach)\b/gi;
    70→   124→  while ((m = formRe.exec(text)) !== null) {
    71→   125→    const windowStart = Math.max(0, m.index - 70);
    72→   126→    const window = text.slice(windowStart, m.index + m[0].length + 70).toLowerCase();
    73→   127→    // ONLY explicit age expressions are exempt ("a 2 months old baby",
    74→   128→    // "5 saal ka bacha") — regimens like "2 tablets every 4 hours" are doses
    75→   129→    if (/ors|oral rehydration|sachet|packet|litre|liter/.test(window)) continue;
    76→   130→    if (/\d+\s*(?:years?|months?|weeks?|days?|saal|mahinay|hafte)\s*old|\b\d+\s*(?:saal|mahinay)\s*(?:ka|ki)\b/.test(window)) continue;

    85→   139→    return true;
